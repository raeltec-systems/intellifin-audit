#!/usr/bin/env bash
set -euo pipefail
. /workspace/zobba-build-tools/activate-tests.sh
cd /workspace/intellifin-audit/zobba
export ZOBBA_TEST_MIGRATION_DATABASE_URL=postgresql://zobba_local_admin@127.0.0.1:55434/zobba_story_20_test
export ZOBBA_TEST_RUNTIME_DATABASE_URL=postgresql://zobba_app@127.0.0.1:55434/zobba_story_20_test
export ZOBBA_TEST_ADMIN_DATABASE_URL=postgresql://zobba_local_admin@127.0.0.1:55434/zobba_story_20_test
export ZOBBA_FIXTURE_DIR=/tmp/zobba-admin-continuity/browser-fixture
export ZOBBA_BROWSER_EXECUTABLE=/usr/bin/chromium
export ZOBBA_BROWSER_OUTPUT_DIR=/tmp/zobba-admin-continuity/review-repair-browser-verified-results
# This is an explicitly disposable test database. Preserve the separate
# zobba_story_20 development database and its running IdP on port 9443.
psql "$ZOBBA_TEST_MIGRATION_DATABASE_URL" -X -v ON_ERROR_STOP=1 <<'SQL' > /tmp/zobba-admin-continuity/review-repair-browser-verified-reset.txt 2>&1
DO $$ BEGIN IF current_database()<>'zobba_story_20_test' THEN RAISE EXCEPTION 'wrong disposable target'; END IF; END $$;
DROP SCHEMA public CASCADE;
CREATE SCHEMA public;
REVOKE CREATE ON SCHEMA public FROM PUBLIC;
SQL
printf '0\n' > /tmp/zobba-admin-continuity/review-repair-browser-verified-reset.exit
result=0
pnpm test:browser --workers=1 --retries=0 > /tmp/zobba-admin-continuity/review-repair-browser-verified.txt 2>&1 || result=$?
printf '%s\n' "$result" > /tmp/zobba-admin-continuity/review-repair-browser-verified.exit
exit "$result"
